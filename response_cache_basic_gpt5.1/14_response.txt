  private static void addFields(ClassNode cls, JsonClsMapping jsonCls) {
      List<FieldNode> fields = cls.getFields();
      if (fields.isEmpty()) {
          return;
      }
      List<JsonFieldMapping> jsonFields = new ArrayList<>(fields.size());
      for (FieldNode field : fields) {

      }
