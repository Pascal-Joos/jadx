		}

		is.skip(4);

		int strIndex = is.readInt32();
		String str = getString(strIndex);
		if (!isLastEnd) {
			isLastEnd = true;
			writer.add('>');
		}

		String escapedStr = StringUtils.escapeXML(str);

		long size = is.readInt16();

}
